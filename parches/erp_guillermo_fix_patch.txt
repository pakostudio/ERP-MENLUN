<script>
/* SMSIDEBARLOGV1: sidebar real para Logistica + ocultar duplicados del topbar */
(function(){
  function esLogistica(){ return typeof currentRole !== 'undefined' && currentRole === 'logistica'; }

  function ocultarTopbarDuplicados(){
    if (!esLogistica()) return;
    ['tbLogisticaBtn','tbPedidosRecibirBtn'].forEach(function(id){
      var el = document.getElementById(id);
      if (el) el.style.display = 'none';
    });
  }

  function agregarItems(){
    if (!esLogistica()) return;
    ocultarTopbarDuplicados();
    if (document.getElementById('smSideLogLogistica')) return;
    var inicio = Array.from(document.querySelectorAll('.nav-item')).find(function(el){ return el.textContent.trim() === 'Inicio'; });
    if (!inicio) return;
    var padre = inicio.parentElement;
    if (!padre) return;

    var ITEMS = [
      { id: 'smSideLogLogistica', label: 'Logística', trigger: 'tbLogisticaBtn' },
      { id: 'smSideLogPedidos', label: 'Pedidos por recibir', trigger: 'tbPedidosRecibirBtn' }
    ];
    var referencia = inicio;
    ITEMS.forEach(function(item){
      var nodo = inicio.cloneNode(true);
      nodo.id = item.id;
      nodo.style.opacity = '1';
      nodo.style.cursor = 'pointer';
      nodo.removeAttribute('data-page');
      nodo.onclick = null;
      nodo.removeAttribute('onclick');
      nodo.classList.remove('active');
      Array.from(nodo.querySelectorAll('*')).forEach(function(el){
        el.onclick = null;
        el.removeAttribute('onclick');
        el.removeAttribute('data-page');
      });
      var textNode = Array.from(nodo.querySelectorAll('*')).find(function(el){ return el.children.length === 0 && el.textContent.trim().length > 0; }) || nodo;
      textNode.textContent = item.label;
      nodo.querySelectorAll('a').forEach(function(a){ a.removeAttribute('href'); });
      nodo.addEventListener('click', function(e){
        e.preventDefault(); e.stopPropagation();
        var trig = document.getElementById(item.trigger);
        if (trig) trig.click();
      }, true);
      padre.insertBefore(nodo, referencia.nextSibling);
      referencia = nodo;
    });
  }

  document.addEventListener('DOMContentLoaded', agregarItems);
  setInterval(agregarItems, 500);
})();
</script>
