<script>
/* SMULTIMACONEXIONV1: panel de ultima conexion por usuario, visible solo para Direccion/Admin, junto al saludo de Inicio */
(function(){
  var SB_URL = 'https://diqbmyqvuyollvlvjniz.supabase.co';
  var SB_KEY = 'sb_publishable_R_XGqN5PlehELFz7u3fgeg_ekGPb38Z';

  function puedeVer(){
    return typeof currentRole !== 'undefined' && ['direccion','admin_sistemas'].indexOf(currentRole) !== -1;
  }

  function formatoRelativo(fechaIso){
    if (!fechaIso) return 'Nunca ha entrado';
    var f = new Date(fechaIso);
    var ahora = new Date();
    var diffMs = ahora - f;
    var diffMin = Math.floor(diffMs / 60000);
    var rel = '';
    if (diffMin < 1) rel = 'hace un momento';
    else if (diffMin < 60) rel = 'hace ' + diffMin + ' min';
    else if (diffMin < 1440) rel = 'hace ' + Math.floor(diffMin/60) + ' h';
    else rel = 'hace ' + Math.floor(diffMin/1440) + ' d';
    var fechaTxt = f.toLocaleString('es-MX', { day:'2-digit', month:'short', year:'numeric', hour:'2-digit', minute:'2-digit' });
    return fechaTxt + ' (' + rel + ')';
  }

  async function cargarPanel(){
    if (!puedeVer()) return;
    if (document.getElementById('smUltimaConexionPanel')) { await refrescar(); return; }

    var hello = document.querySelector('.hello');
    if (!hello) return;
    var contenedor = hello.parentElement;
    if (!contenedor) return;

    var panel = document.createElement('div');
    panel.id = 'smUltimaConexionPanel';
    panel.style.cssText = 'margin:14px 0;padding:12px 16px;border:1px solid #e2e2e2;border-radius:10px;background:#fafafa;font-size:13px;max-width:520px;';
    panel.innerHTML = '<div style="display:flex;align-items:center;justify-content:space-between;cursor:pointer;font-weight:600;color:#333;" id="smUltimaConexionToggle">'
      + '<span>👥 Última conexión por usuario</span><span id="smUltimaConexionCaret">▾</span></div>'
      + '<div id="smUltimaConexionBody" style="margin-top:8px;"></div>';

    var refHello = document.querySelector('.hello-sub') || hello;
    refHello.parentElement.insertBefore(panel, refHello.nextSibling);

    var toggle = document.getElementById('smUltimaConexionToggle');
    var body = document.getElementById('smUltimaConexionBody');
    toggle.addEventListener('click', function(){
      var visible = body.style.display !== 'none';
      body.style.display = visible ? 'none' : '';
      document.getElementById('smUltimaConexionCaret').textContent = visible ? '▸' : '▾';
    });

    await refrescar();
  }

  async function refrescar(){
    var body = document.getElementById('smUltimaConexionBody');
    if (!body) return;
    body.innerHTML = '<div style="color:#888;">Cargando…</div>';
    try {
      var res = await fetch(SB_URL + '/rest/v1/menlun_ultima_conexion?select=*&order=last_sign_in_at.desc.nullslast', {
        headers: { apikey: SB_KEY, Authorization: 'Bearer ' + SB_KEY }
      });
      if (!res.ok) { body.innerHTML = '<div style="color:#c00;">No se pudo cargar.</div>'; return; }
      var rows = await res.json();
      var html = '<table style="width:100%;border-collapse:collapse;">';
      rows.forEach(function(r){
        var activo = r.last_sign_in_at && (new Date() - new Date(r.last_sign_in_at)) < (15 * 60000);
        html += '<tr>'
          + '<td style="padding:3px 6px 3px 0;white-space:nowrap;">' + (activo ? '🟢 ' : '⚪ ') + r.nombre + '</td>'
          + '<td style="padding:3px 0;color:#555;">' + formatoRelativo(r.last_sign_in_at) + '</td>'
          + '</tr>';
      });
      html += '</table>';
      body.innerHTML = html;
    } catch(e) {
      body.innerHTML = '<div style="color:#c00;">Error al cargar.</div>';
    }
  }

  document.addEventListener('DOMContentLoaded', cargarPanel);
  setInterval(cargarPanel, 2000);
})();
</script>
